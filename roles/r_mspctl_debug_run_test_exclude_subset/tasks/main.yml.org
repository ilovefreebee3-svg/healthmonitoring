---
- name: 0. Declare variables
  set_fact:
    read_from_file: false 

- name: 1. Run mspctl debug run --test_exclude_subset
  ansible.builtin.command:
    argv:
      - /bin/bash
      - -c
      - |
        source /etc/profile
        mspctl debug run --test_exclude_subset=hosting_pe
  register: input_filter
  changed_when: false
  failed_when: false
  when: not read_from_file | bool

#- name: 2. Execute a remote script
#  ansible.builtin.command: name_of_remote_script
#  register: input_filter

#  ignore_errors: true

- name: 3. Read from hardcoded file
  set_fact:
    input_filter:
      stdout: "{{ lookup('file', 'data.32_testing11_mspctl_debug_run_test_exclude_subset_site_data.txt') }}"
      stdout_lines: "{{ lookup('file', 'data.32_testing11_mspctl_debug_run_test_exclude_subset_site_data.txt').splitlines() }}"
      rc: 0
      failed: false
      changed: false
  when: read_from_file | bool

- name: 4. Display input_filter.stdout_lines
  ansible.builtin.debug:
    var: input_filter.stdout_lines

- name: 5. Display input_filter.stdout
  ansible.builtin.debug:
    var: input_filter.stdout

- name: 6. Remove ANSI color codes
  ansible.builtin.set_fact:
    Xinput_filter_msg: >-
      {{
        input_filter.stdout_lines
        | map('regex_replace', '\x1B\[[0-9;]*[A-Za-z]', '')
        | list
      }}

- name: 7. Display cleaned output
  ansible.builtin.debug:
    var: Xinput_filter_msg

- name: 8. Display input_filter.stdout_lines
  ansible.builtin.debug:
    msg:
      - "V: {{ input_filter.stdout_lines }}"

- name: 9. Display Xinput_Result msg
  ansible.builtin.debug:
    msg:
      - "X: {{ Xinput_filter_msg }}"

- name: 10. Initialize variables
  set_fact:
    current_name: ""
    failures: []
    test_results: []

- name: 11. Parse in ONE loop (correct pairing)
  set_fact:
    current_name: >-
      {{
      subset_line.split(':', 1)[1] | trim
      if subset_line is search('^Test Name:')
      else current_name
      }}
    test_results: >-
      {{
      test_results + [ {'name': current_name, 'status': subset_line.split(':',1)[1] | trim} ]
      if subset_line is search('^STATUS:')
      else test_results
      }}
    failures: >-
      {{
      failures + [ {'name': current_name, 'status': subset_line.split(':',1)[1] | trim} ]
      if (subset_line is search('^STATUS:') and (subset_line.split(':',1)[1] | trim) != 'SUCCESS')
      else failures
      }}
  loop: "{{ Xinput_filter_msg }}"
  loop_control:
    loop_var: subset_line

- name: 12. Compute summary
  set_fact:
    num_failure: "{{ failures | length }}"
    remark: >-
      {{
        failures
        | map(attribute='name')
        | zip(failures | map(attribute='status'))
        | map('join', ' -> ')
        | join(' -- ')
      }}
    answer: "{{ 'STATUS Not Success' if (failures | length) > 0 else 'STATUS SUCCESS' }}"

- debug:
    msg:
      - "Failures: {{ num_failure }}"
      - "Remark: {{ remark }}"
      - "Answer: {{ answer }}"
