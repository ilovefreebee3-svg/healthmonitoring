# name: Create a simple text file
- name: Ensure /tmp/hello_world.txt exists with content
  ansible.builtin.copy: # Using the copy module
    content: "Hello from your first Ansible playbook!\n"
    dest: /home/user1/proj/ansible/temp/hello_world.txt
    mode: '0644' # Set file permissions

- name: Capture file content into a variable
  set_fact:
    input_text: "{{ lookup('file', 'ncc_health_checks_run_all.txt') }}"

- name: Extract Fail matches
  set_fact:
    fail_matches: "{{ input_text | regex_findall('\\|\\s*Fail\\s*\\|\\s*(\\d+)\\s*\\|') }}"

- name: Determine even/odd
  set_fact:
    answer: "{{
        'Success' if (fail_matches | length == 0)
        else ('Failed' if (fail_matches[0] | int > 0) else 'Success')
      }}"
    remark: "{{ fail_matches[0] }}"

- name: Include common role
  ansible.builtin.include_role:
     name: common
