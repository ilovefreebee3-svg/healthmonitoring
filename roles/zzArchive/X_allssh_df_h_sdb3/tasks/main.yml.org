---
- name: Capture file content into a variable
  set_fact:
    input_text: "{{ lookup('file', 'allssh_df_h_sdb3.txt') }}"


- name: Check if /dev/sdb3 exists anywhere
  set_fact:
    sdb3_lines: "{{ input_text.split('\n') | select('search', '^/dev/sdb3') | list }}"

- name: Exit if /dev/sdb3 not found
  set_fact:
    answer: "/dev/sdb3 not found"
    remark: "This filesystem is not found"
  when: sdb3_lines | length == 0

- meta: end_play
  when: sdb3_lines | length == 0

- name: Initialise variables
  set_fact:
    num_Over75: 0
    remark: ""
    answer: ""
    current_ip: ""

- name: Process all lines in order
  set_fact:
    current_ip: >-
      {{ item | regex_search('=+\s*(\S+)\s*=+', '\1')
         if item is search('^=+') else current_ip }}

    num_Over75: >-
      {{ (num_Over75 | int) +
         (1 if (item is search('^/dev/sdb3') and (use_percent | int > 75)) else 0) }}

    remark: >-
      {{ remark +
         (current_ip ~ ' ' ~ fs ~ ' ' ~ use_percent ~ "  "
         if (item is search('^/dev/sdb3') and (use_percent | int > 75)) else '') }}
  vars:
    parts: "{{ item.split() }}"
    fs: "{{ parts[0] if item is search('^/dev/sdb3') else '' }}"
    use_percent: >-
      {{ (parts[4] | regex_replace('%',''))
         if item is search('^/dev/sdb3') else '0' }}
  loop: "{{ input_text.split('\n') }}"

- name: Final result
  set_fact:
    answer: "{{ 'Over 75%' if (num_Over75 | int) > 0 else 'All OK' }}"

- name: Show answer
  debug:
    msg:
      answer: "{{ answer }}"

- name: Show remark
  debug:
    msg:
      answer: "{{ remark }}"

- name: Include common role
  ansible.builtin.include_role:
     name: common
