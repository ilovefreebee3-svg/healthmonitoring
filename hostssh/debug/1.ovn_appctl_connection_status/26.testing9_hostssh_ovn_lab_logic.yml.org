---
- name: 1. Run diagnostic commands on all CVMs
  hosts: corecvm
#  hosts: engrcvm
#  hosts: pecvm
#  hosts: allcvms
  gather_facts: false
  serial: 1

  vars:
    cmd_list:
      - name: hostssh ovn-appctl connection-status
        argv: ["hostssh", "ovn-appctl", "connection-status"]

#      - name: hostssh_ovn_appctl_connection_status
#        argv: ["hostssh_ovn_appctl_connection_status"]

  tasks:
    - name: 2. Execute commands
      include_tasks: subr.26.testing9_hostssh_ovn_lab_logic.yml
      loop: "{{ cmd_list }}"
      loop_control:
        loop_var: cmd

    - name: 3. Display hostssh sudo ovn_appctl connection-status output
      ansible.builtin.debug:
        msg: "{{ input_filter }}" 

    - name: 4. Split lines
      set_fact:
        lines: "{{ (input_filter.stdout).split('\n') | map('trim') | reject('equalto','') | list }}"

    - name: 5. Initialize results
      set_fact:
        num_disconnected: 0
        remark: ""

    - name: 6. Process IP entries safely
      set_fact:
        num_disconnected: "{{ (num_disconnected | default(0) | int) + (1 if status != 'connected' else 0) }}"
        remark: >-
          {{
            remark +
            (ip ~ ' - ' ~ status ~ '; ' if status != 'connected' else '')
          }}
      loop: "{{ range(0, lines | length) | list }}"
      vars:
        current: "{{ lines[item] }}"
        next_line: "{{ lines[item + 1] if item + 1 < (lines | length) else '' }}"
        is_ip: "{{ current is search('=+\\s*[0-9.]+\\s*=+') }}"
        next_is_ip: "{{ next_line is search('=+') }}"
        ip: "{{ current | regex_search('([0-9.]+)') }}"
        status: >-
          {{
            (next_line if not next_is_ip else '')
          }}
      when: is_ip

    - name: Final answer
      set_fact:
        answer: "{{ 'Disconnected' if (num_disconnected | int) > 0 else 'Connected' }}"

    - debug:
        msg:
          num_disconnected: "{{ num_disconnected }}"
          remark: "{{ remark }}"
          answer: "{{ answer }}"

    - name: Append result to CSV (local)
      import_tasks: logd.yml
